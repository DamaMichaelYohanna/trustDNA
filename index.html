server.js 
import { TrustDNA } from "@trustdna/node";

const trustdna = new TrustDNA({
  secretKey: process.env.TRUSTDNA_SECRET_KEY
});

app.post("/api/transfer", async (req, res) => {
  // Query real-time risk decision from TrustDNA
  const assessment = await trustdna.risk.evaluate({
    userId: req.user.id,
    telemetryToken: req.body.trustdna_token,
    transaction: {
      amount: req.body.amount,
      currency: "NGN"
    }
  });

  // Act on decision based on your configured policies
  if (assessment.decision === "block") {
    return res.status(403).json({ error: "Transaction blocked for security." });
  }

  if (assessment.decision === "challenge") {
    return res.json({ requireMFA: true, challengeId: assessment.challenge_id });
  }

  // Allow transaction to proceed smoothly
  await processPayment(req.body);
  return res.json({ success: true });
});
main.py 
from fastapi import FastAPI, HTTPException
from trustdna import TrustDNAClient

client = TrustDNAClient(secret_key="td_sec_live_9a8f4c2e11")
app = FastAPI()

@app.post("/api/transfer")
async def handle_transfer(payload: TransferRequest):
    # Evaluate risk in < 40ms
    assessment = await client.risk.evaluate(
        user_id=payload.user_id,
        telemetry_token=payload.trustdna_token,
        transaction={"amount": payload.amount, "currency": "NGN"}
    )

    if assessment.decision == "block":
        raise HTTPException(status_code=403, detail="High risk anomaly detected")
    
    if assessment.decision == "challenge":
        return {"step_up_mfa_required": True, "reasons": assessment.reasons}

    # Execute transfer
    return {"status": "approved", "trust_score": assessment.trust_score}

webhooks/trustdna.js 
// Handle asynchronous security events sent from TrustDNA
app.post("/webhooks/trustdna", async (req, res) => {
  const event = req.body;

  switch (event.type) {
    case "account_takeover_detected":
      // Force terminate active session & notify user
      await revokeUserSessions(event.user_id);
      await sendSecurityAlertEmail(event.user_id, event.reasons);
      break;

    case "impossible_travel_flagged":
      await promptSecondaryBiometrics(event.user_id);
      break;
  }

  res.sendStatus(200);
});
